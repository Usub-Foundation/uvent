find_package(Threads REQUIRED)

function(uvent_add_test name)
    add_executable(${name} ${name}.cpp)
    set(UVENT_TEST_TARGETS ${UVENT_TEST_TARGETS} ${name} PARENT_SCOPE)
    target_link_libraries(${name} PRIVATE uvent Threads::Threads)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/include)
    add_test(NAME ${name} COMMAND ${name})
    if (UVENT_TESTS_SANITIZER)
        target_compile_options(${name} PRIVATE -fsanitize=${UVENT_TESTS_SANITIZER} -fno-omit-frame-pointer -g)
        target_link_options(${name} PRIVATE -fsanitize=${UVENT_TESTS_SANITIZER})
    endif ()
endfunction()
